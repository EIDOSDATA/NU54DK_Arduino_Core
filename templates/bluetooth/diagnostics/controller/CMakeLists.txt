cmake_minimum_required(VERSION 3.20.0)
set(DIAGNOSTIC_TRANSPORT "uart" CACHE STRING "uart, async, threewire, lpuart or spi")
if(DIAGNOSTIC_TRANSPORT STREQUAL "uart")
    set(upstream_relative "samples/bluetooth/hci_uart")
elseif(DIAGNOSTIC_TRANSPORT STREQUAL "async")
    set(upstream_relative "samples/bluetooth/hci_uart_async")
elseif(DIAGNOSTIC_TRANSPORT STREQUAL "threewire")
    set(upstream_relative "samples/bluetooth/hci_uart_3wire")
elseif(DIAGNOSTIC_TRANSPORT STREQUAL "lpuart")
    set(upstream_relative "samples/bluetooth/hci_uart")
elseif(DIAGNOSTIC_TRANSPORT STREQUAL "spi")
    set(upstream_relative "samples/bluetooth/hci_spi")
else()
    message(FATAL_ERROR "Unknown or unavailable diagnostic transport")
endif()
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(nucode_raw_controller)
set(upstream_main "${ZEPHYR_BASE}/${upstream_relative}/src/main.c")
set_source_files_properties("${upstream_main}" PROPERTIES COMPILE_DEFINITIONS "main=nucode_upstream_main")
target_sources(app PRIVATE src/main.c "${upstream_main}")
if(DIAGNOSTIC_TRANSPORT STREQUAL "async")
    target_sources(app PRIVATE "${ZEPHYR_BASE}/${upstream_relative}/src/hci_uart_async.c")
endif()
